regionNames = new Intl.DisplayNames(["en"], { type: "region" })

customNames = ({
  "gb-eng": "England",
  "gb-nir": "Northern Ireland",
  "gb-sct": "Scotland",
  "gb-wls": "Wales"
})

getName = (file) => {
  const code = file.replace(".png", "");
  return customNames[code] ?? regionNames.of(code.toUpperCase()) ?? code;
}

// Build a sorted list of { name, file } objects for the dropdown
flags = flag_files
  .map(file => ({ name: getName(file), file }))
  .sort((a, b) => a.name.localeCompare(b.name))

// Dropdown menu for selecting a flag
viewof selected = Inputs.select(flags, {
  label: "Choose a flag:",
  format: d => d.name,
  value: flags.find(d => d.file === "us.png")
})

// Look up the 3 flags with the most similar color and design distributions
similarFlags = similar_map[selected.file].map(file => ({
  file,
  name: getName(file)
}))

// Display the selected flag and its 3 closest color + design matches
html`<div>
  <figure style="margin-top: 1.5rem;">
    <img src="w2560/${selected.file}" alt="${selected.name} flag" style="max-width: 500px; width: 100%; border: 1px solid #ccc; border-radius: 4px;">
    <figcaption style="margin-top: 0.5rem; font-weight: bold; font-size: 1.1rem;">${selected.name} (${selected.file})</figcaption>
  </figure>

  <h3 style="margin-top: 2rem;">3 Flags with Similar Color &amp; Design</h3>
  <div style="display: grid; grid-template-columns: repeat(3, 1fr); gap: 1rem; margin-top: 1rem;">
    ${similarFlags.map(f => html`
      <figure style="margin: 0; text-align: center;">
        <img src="w2560/${f.file}" alt="${f.name} flag" style="width: 100%; height: 140px; object-fit: contain; background: #f8f9fa; border: 1px solid #ccc; border-radius: 4px; padding: 4px;">
        <figcaption style="margin-top: 0.4rem; font-size: 0.95rem;"><strong>${f.name}</strong><br><span style="color: #666; font-size: 0.85rem;">${f.file}</span></figcaption>
      </figure>
    `)}
  </div>
</div>`
