regionNames = new Intl.DisplayNames(["en"], { type: "region" })

customNames = ({
  "gb-eng": "England",
  "gb-nir": "Northern Ireland",
  "gb-sct": "Scotland",
  "gb-wls": "Wales"
})

getName = (file) => {
  const code = file.replace(".png", "");
  return customNames[code] ?? regionNames.of(code.toUpperCase()) ?? code;
}

// Build a sorted list of { name, file } objects for the dropdown
flags = flag_files
  .map(file => ({ name: getName(file), file }))
  .sort((a, b) => a.name.localeCompare(b.name))

// Dropdown menu for selecting a flag
viewof selected = Inputs.select(flags, {
  label: "🔍 Choose a country or territory:",
  format: d => d.name,
  value: flags.find(d => d.file === "us.png")
})

// Helper to jump to any flag when a card or the random button is clicked
selectFlag = (file) => {
  const target = flags.find(d => d.file === file);
  if (target) {
    viewof selected.value = target;
    viewof selected.dispatchEvent(new Event("input", { bubbles: true }));
  }
}

// Surprise Me button to pick a random flag
{
  const btn = html`<button class="random-btn" type="button">🎲 Surprise Me with a Random Flag!</button>`;
  btn.onclick = () => {
    const randomItem = flags[Math.floor(Math.random() * flags.length)];
    selectFlag(randomItem.file);
  };
  return html`<div style="text-align: center; margin-bottom: 1.5rem;">${btn}</div>`;
}
